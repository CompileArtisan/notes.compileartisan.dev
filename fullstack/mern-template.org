* How Authentication works
When you open http://localhost:3000:
- React loads.
- It shows a login or register modal.
- When you register → React sends your info to http://localhost:8080/auth/register.
- Express stores your user in MongoDB.
- When you log in → Express sends back a JWT token.
- React stores that token → and now you’re “logged in”.

* How to add something to this
1. Make mongoose model (mern-template/server/src/models/something.model.ts) 
2. Make an Express route (how to get/send that data), at mern-template/server/src/routes/note.routes.ts 
3. Register the Route in the server at mern-template/server/src/index.ts
4. Make frontend API file mern-template/client/src/api/notes.ts 
5. Make React App mern-template/client/src/components/NotesPage.tsx (which can be used in mern-template/client/src/App.tsx)

** Make a Mongoose database model (what kind of data you store), at ~mern-template/server/src/models/note.model.ts~
For example:
#+begin_src typescript
import mongoose from "mongoose";

const noteSchema = new mongoose.Schema(
  {
    title: { type: String, required: true },
    content: { type: String, required: true },
    userId: { type: mongoose.Schema.Types.ObjectId, ref: "User", required: true },
  },
  { timestamps: true }
);

export default mongoose.model("Note", noteSchema);

#+end_src
This would look like:
#+begin_src txt
MongoDB "notes" collection
 ├─ { title: "Buy milk", content: "At 5 PM", userId: 642ad... }
 ├─ { title: "Workout", content: "Leg day", userId: 642ad... }

#+end_src

** Make an Express route (how to get/send that data), at ~mern-template/server/src/routes/note.routes.ts~
#+begin_src typescript
import { Router } from "express";
import Note from "../models/note.model";

const router = Router();

// GET /notes → fetch all notes for a user
router.get("/", async (req, res) => {
  const { userId } = req.query; // later this comes from the logged-in user
  const notes = await Note.find({ userId });
  res.json(notes);
});

// POST /notes → create a new note
router.post("/", async (req, res) => {
  const { title, content, userId } = req.body;
  const note = await Note.create({ title, content, userId });
  res.status(201).json(note);
});

// DELETE /notes/:id → delete a note
router.delete("/:id", async (req, res) => {
  await Note.findByIdAndDelete(req.params.id);
  res.json({ message: "Deleted" });
});

export default router;

#+end_src

** Make frontend API file ~mern-template/client/src/api/notes.ts~
#+begin_src typescript
import axios from "axios";

const API_URL = import.meta.env.VITE_BACKEND_URL || "http://localhost:8080";

export const getNotes = async (userId: string) => {
  const res = await axios.get(`${API_URL}/notes`, { params: { userId } });
  return res.data;
};

export const addNote = async (note: { title: string; content: string; userId: string }) => {
  const res = await axios.post(`${API_URL}/notes`, note);
  return res.data;
};

export const deleteNote = async (id: string) => {
  await axios.delete(`${API_URL}/notes/${id}`);
};

#+end_src

** Make React App ~mern-template/client/src/components/NotesPage.tsx~ (which can be used in ~mern-template/client/src/App.tsx~)
#+begin_src typescript
import { useEffect, useState } from "react";
import { getNotes, addNote, deleteNote } from "../api/notes";

export default function NotesPage() {
  const [notes, setNotes] = useState<any[]>([]);
  const [title, setTitle] = useState("");
  const [content, setContent] = useState("");

  const userId = "123"; // later: from AuthContext

  // fetch notes when page loads
  useEffect(() => {
    getNotes(userId).then(setNotes);
  }, []);

  const handleAdd = async () => {
    const newNote = await addNote({ title, content, userId });
    setNotes([...notes, newNote]);
    setTitle("");
    setContent("");
  };

  const handleDelete = async (id: string) => {
    await deleteNote(id);
    setNotes(notes.filter((n) => n._id !== id));
  };

  return (
    <div className="p-4">
      <h2 className="text-xl font-bold">My Notes</h2>

      <div className="flex gap-2 my-3">
        <input
          placeholder="Title"
          className="border p-2"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
        />
        <input
          placeholder="Content"
          className="border p-2"
          value={content}
          onChange={(e) => setContent(e.target.value)}
        />
        <button onClick={handleAdd} className="bg-blue-500 text-white px-4 py-2 rounded">
          Add
        </button>
      </div>

      <ul>
        {notes.map((n) => (
          <li key={n._id} className="flex justify-between items-center border-b py-2">
            <div>
              <strong>{n.title}</strong>: {n.content}
            </div>
            <button onClick={() => handleDelete(n._id)} className="text-red-500">
              ✕
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}
